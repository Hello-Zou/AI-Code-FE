创建自定义倒计时

使用方法 

// 创建倒计时实例
const countdown = new Countdown({
    'targetDate': '2026-12-31T23:59:59', // 目标时间
    'format': 'full', // 显示格式
    'updateInterval': 1000, // 更新间隔(毫秒)
    'onUpdate': function(time) {
        // 每次更新时的回调
        console.log(time.days + '天' + time.hours + '时');
    },
    'onComplete': function() {
        // 倒计时结束时的回调
        alert('倒计时结束!');
    }
});

// 开始倒计时
countdown.start();

// 控制方法
countdown.pause();     // 暂停
countdown.resume();    // 继续
countdown.reset();     // 重置
countdown.addTime(3600); // 增加时间(秒)
